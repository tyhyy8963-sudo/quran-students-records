namespace App\Http\Controllers\Auth;

class RegisterController extends Controller
{
    /**
     * عرض صفحة التسجيل
     */
    public function create()
    {
        // نعرض الفورم من welcome.blade.php
        return view('welcome');
    }

    /**
     * حفظ بيانات المستخدم بعد التحقق
     */
    public function store(Request $request)
    {
        // التحقق من صحة البيانات
        $request->validate([
            'name'      => 'required|string|max:255',
            'mosque'    => 'required|string|max:255',

            'mosque'    => $request->mosque,

            'password'  => Hash::make($request->password),
        ]);

        return redirect()->route('dashbord')->with('success', 'تم إنشاء الحساب بنجاح');

<?php

namespace App\Http\Controllers;

use App\Models\Student;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class StudentController extends Controller
{
    public function index()
    {
        $students = Student::with('data')
            ->where('teacher_id', auth()->id())
            ->get();

        return view('dashbord', compact('students'));
    }

    // إنشاء طالب
    public function create(Request $request)
    {
        $validated = $request->validate([
            'student_name' => 'required|string|max:255',
            'the_surah'    => 'nullable|string|max:255',
            'the_ayah'     => 'nullable|integer',
        ]);

        $student = DB::transaction(function () use ($validated) {

            $student = Student::create([
                'student_name' => $validated['student_name'],
                'teacher_id'   => auth()->id(),
            ]);

            $student->data()->create([
                'the_surah' => $validated['the_surah'] ?? null,
                'the_ayah'  => $validated['the_ayah'] ?? null,
            ]);

            return $student;
        });

        return response()->json([
            'id' => $student->student_id
        ]);
    }

    // تعديل طالب (يعمل بلا حدود)
    public function update(Request $request, $id)
    {
        $student = Student::with('data')
            ->where('student_id', $id)
            ->where('teacher_id', auth()->id())
            ->firstOrFail();

        $validated = $request->validate([
            'student_name' => 'required|string|max:255',
            'the_surah'    => 'nullable|string|max:255',
            'the_ayah'     => 'nullable|integer',
        ]);

        DB::transaction(function () use ($student, $validated) {

            $student->update([
                'student_name' => $validated['student_name'],
            ]);

            $student->data()->updateOrCreate(
                ['student_id' => $student->student_id],
                [
                    'the_surah' => $validated['the_surah'] ?? null,
                    'the_ayah'  => $validated['the_ayah'] ?? null,
                ]
            );
        });

        return response()->json(['success' => true]);
    }

    // حذف طالب
    public function destroy($id)
    {
        $student = Student::where('student_id', $id)
            ->where('teacher_id', auth()->id())
            ->firstOrFail();

        $student->delete();

        return response()->json(['success' => true]);
    }
}

function bindRow(row) {
    const save = row.querySelector('.save');
    const edit = row.querySelector('.edit');
    const del  = row.querySelector('.delete');

    save?.addEventListener('click', () => showConfirm('هل تريد حفظ البيانات؟', () => saveStudent(row)));
    edit?.addEventListener('click', () => enableEdit(row));
    del?.addEventListener('click', () => showConfirm('هل تريد حذف البيانات؟', () => deleteStudent(row)));
}

function saveStudent(row) {
    const data = {
        student_name: row.querySelector('.name').value,
        the_surah: row.querySelector('.surah').value,
        the_ayah: row.querySelector('.ayah').value,
    };

    const id = row.dataset.id;
    const url = id ? `/dashbord/${id}` : `/dashbord/create`;
    const method = id ? 'PATCH' : 'POST';

    fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json', 'X-CSRF-TOKEN': csrf },
        body: JSON.stringify(data)
    })
    .then(res => res.json())
    .then(res => {
        row.dataset.id = res.id;
        row.querySelector('.edit').disabled = false;
        disableInputs(row);
        updateCount();
    });
}

function enableEdit(row) {
    row.querySelectorAll('input').forEach(i => i.disabled = false);
}

function deleteStudent(row) {
    const id = row.dataset.id;
    if(!id){
        row.remove();
        updateCount();
        return;
    }

    fetch(`/dashbord/${id}`, {
        method: 'DELETE',
        headers: { 'X-CSRF-TOKEN': csrf }
    }).then(() => {
        row.nextElementSibling?.remove();
        row.remove();
        updateCount();
    });
}

function disableInputs(row){
    row.querySelectorAll('input').forEach(i => i.disabled = true);
}

function updateCount(){
    document.getElementById('studentsCount').textContent =
        document.querySelectorAll('.student-row[data-id]').length;
}
